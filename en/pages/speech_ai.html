🐍 Python example
python
import whisper

# Charger le modèle (téléchargé si absent)
model = whisper.load_model('large-v3')

# Transcrire un fichier audio
result = model.transcribe(
    'enregistrement.mp3',
    language='fr',
    verbose=True
)

print(result['text'])

# Avec timestamps
for segment in result['segments']:
    print(f"[{segment['start']:.1f}s] {segment['text']}")
