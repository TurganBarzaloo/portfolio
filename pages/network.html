🔒 Hardening SSH
bash
# /etc/ssh/sshd_config
PermitRootLogin no
PasswordAuthentication no
PubkeyAuthentication yes
MaxAuthTries 3
ClientAliveInterval 300

# Générer une clé SSH
ssh-keygen -t ed25519 -C 'portfolio-ia'

# Copier la clé
ssh-copy-id -i ~/.ssh/id_ed25519.pub user@proxmox-host
