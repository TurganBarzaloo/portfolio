💻 Commandes clés
bash
# Liste des VM
qm list

# Démarrer une VM
qm start <VMID>

# Snapshot avant modification
qm snapshot <VMID> pre-update-$(date +%Y%m%d)

# Backup
vzdump <VMID> --storage local --mode snapshot

# Clone depuis template
qm clone <template_id> <new_id> --name ma-vm --full
