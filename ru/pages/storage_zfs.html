💻 Команды ZFS
bash
# Lister tous les datasets
zfs list

# Créer un dataset
zfs create pool/ollama

# Snapshot avant migration
zfs snapshot pool/ollama@avant-migration-$(date +%Y%m%d)

# Lister les snapshots
zfs list -t snapshot

# Restaurer un snapshot
zfs rollback pool/ollama@avant-migration

# Vérifier santé du pool
zpool status
zpool scrub pool
